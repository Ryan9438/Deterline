  --key KEY             键名或数字索引 (file0 用数字)
  --value VALUE         要设置/搜索的值 (set/find 操作必需)
  --old OLD             replace-string 的旧文本 (decompile-code 兼容别名，等同 --search)
  --search SEARCH       decompile-code: 仅显示匹配该文本的行 (等同行过滤)
  --preview             运行检查并预览拟写入文件，不提交编辑方案
  --dry-run             预览替换结果/匹配数，不写入文件 (replace-code/find-replace-code 用)
  --section SECTION     INI 节名 (默认: General)
  --file {ini,file0,data_win}
                        目标文件 (默认: ini)
  --dir DIR             手动指定存档目录或游戏数据文件路径 (backup-index/recovery-scan 亦接受含
                        game.ios/data.win 的目录)
  --limit LIMIT         最大显示条数 (默认: 30)
  --offset OFFSET       起始偏移 (list-strings 用)
  --hex                 显示十六进制 (list-strings 用)
  --file-path           --value 是文件路径而非内联内容 (replace-code 用)
  --backup BACKUP       备份时间戳 (undo 恢复到指定版本, list-backups 可查看)
  --x X                 X 坐标 (spawn-obj/create-door 用, 默认: 320)
  --y Y                 Y 坐标 (spawn-obj/create-door 用, 默认: 240)
  --spawn-x SPAWN_X     目标房间出生 X (create-door 用, 默认: 160)
  --spawn-y SPAWN_Y     目标房间出生 Y (create-door 用, 默认: 120)
  --border BORDER       战斗边框样式 (create-battle 用, 默认: 0)
  --all                 移除所有匹配实例 (remove-obj 用)
  --instance INSTANCE   实例 ID (remove-obj 用, 优先级高于 --key 匹配)
  --face FACE           头像角色 (0=无, 1=Toriel, 2=Flowey, 3=Sans, 4=Papyrus,
                        5=Undyne, 6=Alphys, 7=Asgore, 8=Mettaton, 9=Asriel)
  --emotion EMOTION     头像表情 (Sans: 0=普通, 1=轻笑, 2=眨眼, 3=慢眨眼, 4=无眼)
  --flag FLAG           flag 索引 (quest-init 用, 默认 510)
  --text-key TEXT_KEY   textdata key (npc-dialogue/guard-npc-alarm/npc-choice
                        用)
  --text TEXT           对话文本 (npc-dialogue 用 ASCII；cjk-trigger 用
                        CJK，触发框全形态+预折行自动补全)
  --yes-text-key YES_TEXT_KEY
                        YES 分支 textdata key (npc-choice 用)
  --yes-text YES_TEXT   YES 分支对话内容 ASCII (npc-choice 用)
  --no-text-key NO_TEXT_KEY
                        NO 分支 textdata key (npc-choice 用, 必需)
  --no-text NO_TEXT     NO 分支对话内容 ASCII (npc-choice 用, 必需)
  --goto GOTO           YES 分支传送目标房间（中文名/ID，npc-choice 用）
  --item ITEM           YES 分支奖励物品 ID (npc-choice 用, 0-65)
  --range RANGE         触发距离像素 (npc-dialogue/npc-choice 用, 默认: 50)
  --repeat              可重复触发（走开再靠近可再触发；npc-choice 用，默认单次）
  --frame FRAME         帧索引 (export-sprite/replace-sprite 用, 默认: replace 全部帧的
                        0 / export 全部帧到目录; export 时 --value 以 .png
                        结尾则默认单帧导出到该文件)
  --show-diff           替换代码时显示新旧 diff (replace-code 用)
  --json                以 JSON 输出结构化结果 (backup-index/recovery-scan/backup-
                        prune 用; export-sprite 为附加元数据)
  --keep-archives KEEP_ARCHIVES
                        backup-prune 每目标保留的最近归档份数 (默认: 3)
  --hue-from HUE_FROM   色相范围 "min-max" (colorize-sprite 用, 如 "180-260")
  --hue-to HUE_TO       目标色相角度 0-360 (colorize-sprite 用)
  --workspace WORKSPACE
                        选择独立实验工作区（读取也使用该工作区）
  --sandbox             沙箱模式: 在实验工作区副本上操作（v2 起写操作默认即沙箱，此参数保留为显式声明）
  --commit              v2 写入安全: 直写真实文件（默认写操作只落沙箱副本）；含 Steam 同步
  --no-steam-sync       v2 写入安全: 即使 --commit 也不同步 Steam 路径
  --allow-cjk           C3a: 放行 CJK 对话文本（需中文字体注入支持，否则显示空白）
  --asm                 verify-code: 输出完整反汇编文本（ASM| 行）
  --work-dir WORK_DIR   cjk-font: 中间产物目录（pass1 dump/图集；默认 game 文件旁 cjk_work/）
  --room ROOM           cjk-trigger: 目标房间名（默认 room_ruins1）
  --object OBJECT       cjk-trigger: 承载触发块的对象名（其 Step_1 被追加；默认 obj_cactus）
  --typer TYPER         cjk-trigger: SCR_TEXTTYPE 编号（默认 31 = fnt_maintext
                        spacing12；须在 CJK_TYPER_TABLE）
  --glyphs GLYPHS       cjk-check/cjk-trigger: 字形表路径（默认
                        build/font_cjk/glyphs.json）
  --detail              diff-snapshots: 输出资源级差异（字符串哈希/实例坐标/代码指纹/字体）
  --expected EXPECTED   diff-snapshots --detail: 声明允许改变的资源范围 JSON（范围外差异退出码 1）
  --utmt-timeout UTMT_TIMEOUT
                        diff-snapshots: 单次 UTMT 遍历超时（默认 300s，--detail
                        因全量反汇编指纹需更长）
  --hash                backup-index: 逐项追加 SHA-256（大文件耗时）
  --no-wrap             cjk-trigger: 关闭预折行（默认开）
  --auto-snapshot       每次写入操作前自动创建实验快照
  --apply               应用模式 (handler-extract 用, 默认仅诊断)
  --once                单次触发（触发后 NPC 自毁消失；不锁移动——状态机形态 v2.3.0 已废弃）
  --timeout TIMEOUT     smoke-test 轮询超时秒数 (默认 120)
  --keys {external,osascript}
                        smoke-test 按键模式: external(默认,由调用方按键)/osascript(盲发
                        Z,需辅助功能权限)
  --fix                 自动修复 (verify-nsid 用: 重置损坏变量的 NameStringID；verify-
                        instances 用: 为 ID=0/重复实例分配唯一 ID)
  --width WIDTH         房间宽度 (create-room/set-room-prop)；cjk-trigger:
                        预折行宽度（房间像素，默认 typer31=202 且不超该 typer 的 writingxend）
  --height HEIGHT       房间高度 (create-room/set-room-prop 用)
  --speed SPEED         房间帧率 (create-room/set-room-prop 用)
  --caption CAPTION     房间标题 (create-room/set-room-prop 用)
  --bg-color BG_COLOR   背景色 hex RGB (create-room/set-room-prop 用, 默认: 000000)
  --persistent          房间是否持久化 (set-room-prop 用)
  --bg BG               背景资源名 (set-room-bg/gml-bg 用; create-bg-asset 的资产名走
                        --value, PNG 走 --image)
  --index INDEX         背景/视图层索引 (set-room-bg/set-room-view 用, 默认: 0)
  --enable ENABLE       启用/禁用 bool (set-room-bg/set-room-view 用)
  --foreground FOREGROUND
                        是否前景层 bool (set-room-bg 用)
  --speedx SPEEDX       背景水平视差速度 (set-room-bg 用)
  --speedy SPEEDY       背景垂直视差速度 (set-room-bg 用)
  --tiled-h TILED_H     水平平铺 bool (set-room-bg 用)
  --tiled-v TILED_V     垂直平铺 bool (set-room-bg 用)
  --stretch STRETCH     拉伸 bool (set-room-bg 用)
  --follow FOLLOW       视角跟随对象 (set-room-view 用, 如 obj_mainchara)
  --view-w VIEW_W       视图宽度 (set-room-view 用)
  --view-h VIEW_H       视图高度 (set-room-view 用)
  --port-w PORT_W       视口宽度 (set-room-view 用)
  --port-h PORT_H       视口高度 (set-room-view 用)
  --border-x BORDER_X   视图跟随水平边界 (set-room-view 用)
  --border-y BORDER_Y   视图跟随垂直边界 (set-room-view 用)
  --image IMAGE         PNG 图片路径 (create-bg-asset 用)
  --slot SLOT           背景槽位索引 (gml-bg 用, 默认: 0)
  --layers LAYERS       背景层索引列表, 逗号分隔 (gml-bg/set-room-solid-bg 用, 如 0,1)
  --all-layers          应用到全部 8 个背景层 (gml-bg 用)
  --hide-above          隐藏目标层之上的所有层, 防止遮挡 (gml-bg 用)
  --keep-bg-color       启用背景层时保留 DrawBackgroundColor (set-room-bg 用)
  --color COLOR         纯色 hex RGB, 6 位 (set-room-solid-bg 用, 如 0000FF)
  --asset-name ASSET_NAME
                        背景资源名 (set-room-solid-bg 用, 默认 bg_solid_<color>)
  --no-runtime          只改房间数据, 不注入运行时覆盖 (set-room-solid-bg 用)
